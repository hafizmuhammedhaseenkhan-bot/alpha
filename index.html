""" اردو چیٹ بوٹ — ایک سادہ AI جو آپ سے اردو میں بات کرے گا """ import google.generativeai as genai import sys # ===== اپنا API Key یہاں لگائیں ===== # مفت API Key حاصل کرنے کے لیے: https://aistudio.google.com/app/apikey API_KEY = "YOUR_GEMINI_API_KEY" genai.configure(api_key=API_KEY) # ماڈل کو اردو میں بات کرنے کی ہدایت system_prompt = """ تم ایک دوستانہ اور مددگار اردو چیٹ بوٹ ہو۔ ہمیشہ اردو میں جواب دو۔ سادہ اور واضح الفاظ استعمال کرو۔ صرف اردو میں جواب دو، اردو اسکرپٹ میں۔ """ model = genai.GenerativeModel( model_name="gemini-1.5-flash", system_instruction=system_prompt, ) chat = model.start_chat() print("=" * 50) print("🤖 اردو چیٹ بوٹ میں خوش آمدید!") print("👋 آپ جو چاہیں مجھ سے پوچھ سکتے ہیں۔") print("❌ 'خدا حافظ' یا 'الوداع' لکھ کر بات ختم کریں۔") print("=" * 50) while True: try: user_input = input("\n👤 آپ: ").strip() if not user_input: continue if user_input.lower() in ["خدا حافظ", "الوداع", "bye", "ختم"]: print("🤖 بوٹ: خدا حافظ! اللہ آپ کو خوش رکھے۔ 👋") break response = chat.send_message(user_input) print(f"🤖 بوٹ: {response.text}") except KeyboardInterrupt: print("\n\n🤖 بوٹ: خدا حافظ!") sys.exit(0) except Exception as e: print(f"⚠️ خرابی: {e}") print("💡 براہِ کرم اپنا انٹرنیٹ کنکشن چیک کریں۔")
