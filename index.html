import pyttsx3 import os import webbrowser import datetime engine = pyttsx3.init() voices = engine.getProperty('voices') engine.setProperty('voice', voices[0].id) def speak(audio): print(f"JARVIS: {audio}") engine.say(audio) engine.runAndWait() if _name_ == "_main_": speak("नमस्ते सर, मैं आपका जार्विस हूँ। बताइए मैं आपकी क्या मदद कर सकता हूँ?") while True: query = input("\nआप यहाँ टाइप करें: ").lower() if 'open youtube' in query or 'यूट्यूब खोलो' in query: webbrowser.open("https://www.youtube.com") speak("यूट्यूब खोला जा रहा है।") elif 'time' in query or 'समय' in query: strTime = datetime.datetime.now().strftime("%H:%M:%S") speak(f"सर, अभी समय {strTime} हुआ है।") elif 'shutdown' in query or 'कंप्यूटर बंद करो' in query: speak("कंप्यूटर बंद किया जा रहा है।") os.system("shutdown /s /t 1") elif 'hello' in query or 'हैलो' in query: speak("हैलो सर! सब कुछ ठीक ठाक है?") elif 'how are you' in query or 'कैसे हो' in query: speak("मैं एकदम ठीक हूँ, आप बताइए आप कैसे हैं?") elif 'exit' in query or 'बंद करो' in query or 'quit' in query: speak("अलविदा सर, आपका दिन शुभ हो!") break else: speak("माफ़ कीजिए, मैं इस कमांड को नहीं समझ पाया। कृपया दूसरा कमांड दें।")
